FILE(GLOB SRCS *.cpp)
FILE(GLOB HDRS *.h)

SOURCE_GROUP("" FILES ${SRCS} ${HDRS})

NL_TARGET_LIB(mesh_utils ${SRCS} ${HDRS})

TARGET_INCLUDE_DIRECTORIES(mesh_utils PRIVATE ${ASSIMP_INCLUDE_DIRS})

# AI_INT64 and AI_UINT32 metadata types are missing from older Assimp (e.g. 5.2.0)
# Directory includes are needed since MSVC builds use /X with explicit VC and SDK include dirs
INCLUDE(CheckCXXSourceCompiles)
GET_DIRECTORY_PROPERTY(_MESH_UTILS_DIR_INCLUDES INCLUDE_DIRECTORIES)
SET(CMAKE_REQUIRED_INCLUDES ${_MESH_UTILS_DIR_INCLUDES} ${ASSIMP_INCLUDE_DIRS})
SET(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
CHECK_CXX_SOURCE_COMPILES("#include <assimp/scene.h>
int main() { return (int)AI_INT64 + (int)AI_UINT32; }" NL_ASSIMP_HAS_META_INT64_UINT32)
UNSET(CMAKE_TRY_COMPILE_TARGET_TYPE)
UNSET(CMAKE_REQUIRED_INCLUDES)
IF(NL_ASSIMP_HAS_META_INT64_UINT32)
  TARGET_COMPILE_DEFINITIONS(mesh_utils PRIVATE NL_ASSIMP_HAS_META_INT64_UINT32)
ENDIF()

TARGET_LINK_LIBRARIES(mesh_utils ${ASSIMP_LIBRARIES} nel_gltf NeL::misc NeL::pipeline NeL::3d)
NL_DEFAULT_PROPS(mesh_utils "NeL, Tools, 3D: Mesh Utils")
NL_ADD_RUNTIME_FLAGS(mesh_utils)

IF(WITH_INSTALL_LIBRARIES OR NOT WITH_STATIC)
  INSTALL(TARGETS mesh_utils LIBRARY DESTINATION ${NL_LIB_PREFIX} ARCHIVE DESTINATION ${NL_LIB_PREFIX} COMPONENT tools3d)
ENDIF()
